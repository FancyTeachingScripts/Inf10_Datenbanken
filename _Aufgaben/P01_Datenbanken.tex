\newcommand{\dbprojekt}[9]
{
\Datum{#1 bis #8}

\Aufgabe{Rahmenbedingungen}{
\begin{itemize}
    \item Ausarbeitung und abschließender Test einer Projektidee im 2er-Team
    \item Erstellung von Umsetzungsplan und Datenbank in 4er-Teams (bestehend aus zwei 2er-Teams)
    \item Melden der Gruppen bei der Lehrkraft vor der ersten Abgabe, sonst ist keine Abgabe möglich!
    \item Entwicklung nach dem Wasserfall-Modell:  \UrlAndCode{bycs.link/inf10-wasserfall} oder \\\UrlAndCode{de.wikipedia.org/wiki/Wasserfallmodell}
    \item Als Datenbanksoftware wird die kostenlose Online-Software \emphColB{SQLution} verwendet (auf jedem Gerät verfügbar unter: \UrlAndCode{sqlution.de}, Accounts bekommt ihr von eurer Lehrkraft).
    \item Zusätzlich zur Arbeit im Informatik-Unterricht wird auch \emphColB{Arbeit zuhause} notwendig sein. \emphColB{Arbeitsteilung mit ähnlichem Aufwand für jede:n ist erwünscht!}
    \item Alle \emphColB{Abgaben auf \ValOrInputField{Artemis}{2cm}{colB}} im PDF-Format (Dokumente) bzw. als ZIP-Datei (Datenbank; Download in SQLution). Bei kurzfristigen technischen Problemen mit der Abgabe bitte per BYCS-Drive freigeben.
\end{itemize}
}

\Aufgabe{Inhaltliche Anforderungen}{
\begin{itemize}
    \item Mindestens 4 inhaltlichen Tabellen (Beziehungstabellen zusätzlich); uni- und bidirektionale Beziehungen Normalform; Bonus-Punkte: Erfüllung der 3. Normalform
    \item Auslesen von Daten durch den Nutzer (zur Erfüllung der Anforderungen) ausschließlich über SQL-Abfragen (mit Eingabefeldern oder Dropdown-Menüs).
\end{itemize}
}

\Aufgabe{Anforderung einzelner Phasen}{
\label{sec:details}
}

\UnterAufgabe{Anforderungsanalyse (2er-Team)}{
\subsubsection*{Anforderungsanalyse (2er-Team)}
\begin{itemize}
    \item Einarbeitung in das Wasserfallmodell (kann in Quizzes abgefragt werden).
    \item Zunächst Entwurf des Projekttitels (für Abstimmung, was 1./2./3. Wahl in der Umsetzung im 4er-Team ist), dann Ausarbeitung detaillierter Anforderungen.
    \item Allgemeine Beschreibung des Produkts und Einsatz (\emphColA{Lastenheft})
    \item mindestens 6 funktionale und 1 nicht-funktionale Anforderung (nummeriert im \emphColA{Lastenheft}):\\
    \UrlAndCode{de.wikipedia.org/wiki/Anforderung_(Informatik)}
\end{itemize}
}

\UnterAufgabe{Datenbankentwurf (4er-Team)}{
\subsubsection*{Datenbankentwurf (4er-Team)}
\begin{itemize}
    \item Einarbeitung in Datenmodellierung (Redundanz, Konsistenz, Anomalien): \UrlAndCode{bycs.link/sql_anomalien} (Quiz!)
    \item Basierend auf eurer Abstimmung für ein Thema wird euch ein Lastenheft (keines, das jemand aus eurer Gruppe erstellt hat) zugewiesen, mit dem ihr ab sofort weiterarbeitet.
    \item Entwurf eines Datenbankschemas als Klassendiagramm in SQLution.
    $\rightarrow$ Klassendiagramm (\emphColA{Pflichtenheft})
    \item Entwurf User Interface (= gewünschte Eingabefelder und Dropdown Menüs + Aufbau Ausgabetabellen der SQL-Abfragen)
    $\rightarrow$ Festhalten mittels Skizzen/Beschreibung (\emphColA{Pflichtenheft})
    \item Beschreibung von Maßnahmen, um Konsistenz sicherzustellen / Anomalien zu vermeiden (\emphColA{Pflichtenheft}).
    \item Wo möglich: Zuordnung der Inhalte zu den Anforderungen aus dem Lastenheft (\emphColA{Pflichtenheft}).
\end{itemize}
}

\UnterAufgabe{Umsetzung (4er-Team)}{
\subsubsection*{Umsetzung (4er-Team)}
\begin{itemize}
    \item Haltet euch möglichst genau ans Pflichtenheft; falls ihr etwas ändern müsst, listet es im Bericht auf.
    \item Bearbeitung möglichst arbeitsteilig (Vorsicht: in SQLution nicht gleichzeitig an der gleichen SQL-Abfrage arbeiten!).
    \item \emphColA{Achtung: } Nach einer Änderung des Datenbankmodells muss eine neue Datenbank generiert und alle Datensätze neu eingetragen werden! Also erst gutes Daten-Design überlegen, dann umsetzen.
\end{itemize}
}

\UnterAufgabe{Tests (2er- und 4er-Team)}{
\subsubsection*{Tests (2er- und 4er-Team)}
\begin{itemize}
    \item \emph{4er-Team:} Überprüfung eurer Datenbank anhand des Pflichtenhefts (dieses Testprotokoll ist \emphColA{Teil eurer Doku}).
    \item \emph{2er-Team:} Überprüfung einer fremden Datenbank anhand des Lastenhefts (falls eine Gruppe euer Lastenheft umgesetzt hat, bekommt ihr dieses, ansonst ein zufälliges zugewiesen) (\emphColA{separates Testprotokoll})
    \item Gesamtbeurteilung, ob das Projekt erfolgreich umgesetzt wurde (\emphColA{Testprotokoll}).
    \item Festhalten aller Ergebnisse in einem Testprotokoll (eine Auflistung, welche Anforderung wann von wem geprüft wurde und ob die Überprüfung erfolgreich war und, wenn nicht, wieso).
\end{itemize}
}

\UnterAufgabe{Dokumentation (4er-Team)}{
\subsubsection*{Dokumentation (4er-Team)}
\begin{itemize}
    \item Inhalt: \emphColB{Lastenheft, Pflichtenheft, Testprotokoll, Bedienungsanleitung, Finales Datenbankschema} (mit allen vorgenommenen Änderungen), \emphColB{alle SQL-Abfragen} (SQL-Code)
    \item Optisch ansprechendes Dokument!
\end{itemize}
}

\UnterAufgabe{Bericht/Reflexion über... (4er-Team)}{
\subsubsection*{Bericht/Reflexion über... (4er-Team)}
\begin{itemize}
    \item ...die einzelnen Projektphasen, aufgetretene Schwierigkeiten und was ihr wieder so/anders machen würdet.
    \item ...während der Umsetzung erfolgte Änderungen gegenüber dem Pflichtenheft.
    \item ...die erfolgte Arbeitsteilung.
\end{itemize}
}
